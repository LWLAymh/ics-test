sem_t s;
int main()
{
int i;
pthread_t tids[3];

for (i=0; i <3; i++) {
    pthread_create(&tids[i], NULL, justdoit, NULL);
}
for (i=0; i <3; i++) {
    pthread_join(&tids[i], NULL);
}
return 0;
}
int j=0;
void *justdoit(void *arg)
{
    P(&s);
    j = j + 1;
    V(&s);
    printf(“%d\n”, j);
}
