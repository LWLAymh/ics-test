第五题（10分）异常
以下程序运行时系统调用全部正确执行，且每个信号都被处理到。请给出代码运行
后所有可能的输出结果。
```
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <signal.h>
#include <sys/wait.h>
int c = 1;
void handler1(int sig) {
  c++;
  printf("%d", c);
}
int main() {
  signal(SIGUSR1, handler1);
  sigset_t s;
  sigemptyset(&s);
  sigaddset(&s, SIGUSR1);
  sigprocmask(SIG_BLOCK, &s, 0);
  int pid = fork() ? fork() : fork();
  if (pid == 0) {
    kill(getppid(), SIGUSR1);
    printf("S");
    sigprocmask(SIG_UNBLOCK, &s, 0);
    exit(0);
  } else {
    while (waitpid(-1, NULL, 0) != -1);
    sigprocmask(SIG_UNBLOCK, &s, 0);
    printf("P");
  }
  return 0;
}
```
