```
long absdiff(long x, long y) absdiff:
{ cmpq %rsi, %rdi
long result;
if (x <= y) movq %rdi, %rax
result = x-y; subq %rsi, %rax
else
result = y-x; .L4:
return result; movq %rsi, %rax
} subq %rdi, %rax
ret
```
