```c
#include <stdio.h>
#include <unistd.h>
#include <sys/wait.h>
int x = 1;
int main() {
    printf("start: x = %d\n", x);
    pid_t pid1 = fork();
    if (pid1 == 0) {
        x += 1;
        printf("x = %d\n", x);
        sleep(1);
        pid_t pid2 = fork();
        if (pid2 == 0) {
            x += 2;
            printf("x = %d\n", x);
        } else {
            x += 3;
            printf("x = %d\n", x);
            wait(NULL);
        }
    } else {
        x += 4;
        printf("x = %d\n", x);
        wait(NULL);
        printf("x = %d\n", x);
    }
    return 0;
}
```
