23.（3分）阅读下面这段程序：

```c
#include <stdio.h>
#include <unistd.h>

int main() {
    printf("A\n");
    pid_t pid = fork();
    if (pid == 0) {
        printf("B\n");
    } else {
        printf("C\n");
        fork();
    }
    printf("D\n");
    return 0;
}
```
