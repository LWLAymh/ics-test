```c
long test(long x, long y, long z) {
    long val = {{blank:init}};

        if ({{blank:c2}})
            val = {{blank:v1}};
        else

    } else if ({{blank:c3}})

    return val;
}
```

GCC 产生如下的汇编代码：

```text
long test(long x, long y, long z)

test:
    leaq  (%rdi,%rsi), %rax
    addq  %rdx, %rax
    cmpq  $-3, %rdi
    jge   .L2
    cmpq  %rdx, %rsi
    jge   .L3
    movq  %rdi, %rax
    imulq %rsi, %rax
    ret
.L3:
    movq  %rsi, %rax
    imulq %rdx, %rax
```

%%% reference
这个题目要求补全一个嵌套的分支结构，大部分情况下机器代码就是 C 代码的直接翻译。

补全后的 C 代码：

```c
long test(long x, long y, long z) {
    long val = x+y+z;
    if (x < -3) {
        if (y < z)
            val = x*y;
        else
            val = y*z;
    } else if (x > 2)
        val = x*z;
    return val;
}
```

推导要点：

- 开头 `leaq (%rdi,%rsi),%rax` 与 `addq %rdx,%rax` 计算 `x + y + z`，就是 `val` 的初值。
- `cmpq $-3,%rdi` / `jge .L2`：当 `x >= -3` 时跳到 `.L2`，说明进入内层分支的条件是 `x < -3`。
- 内层 `cmpq %rdx,%rsi` / `jge .L3`：当 `y >= z` 时跳到 `.L3`，所以 then 分支条件是 `y < z`，对应 `val = x*y`；`.L3` 处是 `movq %rsi,%rax` + `imulq %rdx,%rax`，即 `val = y*z`。
- `.L2` 处 `cmpq $2,%rdi` / `jle .L4`：当 `x <= 2` 时不赋值直接返回，所以 `else if` 的条件是 `x > 2`，对应 `val = x*z`（`movq %rdi,%rax` + `imulq %rdx,%rax`）。
