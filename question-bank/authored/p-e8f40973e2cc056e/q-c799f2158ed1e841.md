```c
// main.c
#include <stdio.h>
extern int sum_array(int *arr, int n);
int arr[5] = {1, 2, 3, 4, 5}, result;
static int fix = 2010;
int main() {
    result = sum_array(arr, 5)+fix;
    printf("sum = %d\n", result);
    return 0;
}
```

```c
// sum.c
int sum_array(int *arr, int n) {
    int sum = 0;
```
