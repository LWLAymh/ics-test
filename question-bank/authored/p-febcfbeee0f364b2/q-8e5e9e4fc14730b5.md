#define CNT 4
int arr_data[CNT][CNT] =
    {{900, 901, 902, 903},
     {910, 911, 912, 913},
     {920, 921, 922, 923},
     {930, 931, 932, 933}};
int get_digit(int x, int y)
{
    return arr_data[x][y];
}
