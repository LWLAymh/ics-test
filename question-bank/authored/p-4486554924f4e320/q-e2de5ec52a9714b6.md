fun:
    pushl %ebp
    movl %esp, %ebp
    pushl %ebx
    movl $0, %ecx
    cmpl $0, 8(%ebp)
    jle .L1
.L2:
    movl $0, %edx
    movl 8(%ebp), %eax

    jne .L3
    movl $1, %eax

.L3:
    movl $0, %eax
.L4:
    popl %ebx
    leave
    ret

int fun(unsigned x) {
    int bit_sum = 0;
    while ((int) x > 0) {
        bit_sum += x % 10;
        x = x / 10;
    }
    if (bit_sum % 3 == 0)
        return 1;
    else
        return 0;
}

%%% part-reference: explanation
对 `0 <= x <= 2^31-1`，代码求十进制各位之和，再判断该和是否为3的倍数，等价于判断x是否为3的倍数；x为0时也返回1。对于 `2^31 <= x <= 2^32-1`，`(int)x` 按本题补码机器模型为负，循环跳过，`bit_sum=0`，故直接返回1，不再正确判断所有这类x的整除性。

原PDF的范围是 $2^{n-1}-1$，n为整数位宽；旧转写漏掉指数中的减一，误成 $2^n-1$。这里按32位明确写出阈值。网页保留四个自动核对的表达式，并为原本漏掉答题框的功能说明单列简答。
