第四题（10分）
在x86_64环境下，考虑如下4个文件（main.c, value.c, f1.c, f2.c）：

本题限定原卷的教学机器级模型：传统ELF/common规则（编译时显式 `-fcommon`），SysV x86-64 ABI，按列出的函数体依次执行，不做跨模块优化；调用void函数后的返回寄存器值视为不确定，填X。源码包含经不兼容函数指针类型的调用及函数/对象指针转换，严格ISO C不保证这里的运行输出；本题不是声称这种写法具有可移植行为。
```
/* main.c */
#include <stdio.h>
extern void f_void_void();
extern int f_int_void();
void *f;
int main()
{
    int a = 1, b, c;
    f = (void *)f_void_void;
    ((void (*)(int))f)(a);
    b = ((int (*)())f)();
    printf("b = %d\n", b);
    f = (void *)f_int_void;
    ((void (*)())f)(a);
    c = ((int (*)(int))f)(a);
    printf("c = %d\n", c);
    return 0;
}
/* value.c */
int BIG;
/* f1.c */
#include <stdio.h>
extern int BIG;
int small = 1;
void f_void_void() {
    small += 1;
    BIG += 1;
    printf("small = %d, BIG = %d\n", small, BIG);
}
```
```
/* f2.c*/
#include <stdio.h>
extern int BIG;
static int small;
int f_int_void() {
    small += 1;
    BIG += 1;
    printf("small = %d, BIG = %d\n", small, BIG);
    return small + 1;
}
```
使用命令
`gcc -fcommon -o main main.c f1.c f2.c value.c`
编译这四个文件。
使用
`./main`
运行编译好的程序。
