第四题（10分）
在x86_64环境下，考虑如下4个文件（main.c, value.c, f1.c, f2.c）：
```
/* main.c */
#include <stdio.h>
extern void f_void_void();
extern int f_int_void();
void *f;
int main()
{
    int a = 1, b, c;
    f = (void *)f_void_void;
    ((void (*)(int))f)(a);
    b = ((int (*)())f)();
    printf("b = %d\n", b);
    f = (void *)f_int_void;
    ((void (*)())f)(a);
    c = ((int (*)(int))f)(a);
    printf("c = %d\n", c);
    return 0;
}
/* value.c */
int BIG;
/* f1.c */
#include <stdio.h>
extern int BIG;
int small = 1;
void f_void_void() {
    small += 1;
    BIG += 1;
    printf("small = %d, BIG = %d\n", small, BIG);
}
```
```
/* f2.c*/
#include <stdio.h>
extern int BIG;
static int small;
int f_int_void() {
    small += 1;
    BIG += 1;
    printf("small = %d, BIG = %d\n", small, BIG);
    return small + 1;
}
```
使用命令
gcc -o main main.c f1.c f2.c value.c
编译这四个文件。
使用
`./main`
运行编译好的程序。
