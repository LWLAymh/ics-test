空，HCL代码中6空）。

    icode in { IRRMOVQ,
    } : valA;
    icode in { IIRMOVQ,
    , IMRMOVQ } : valC;
    icode in {
    ;
    icode in {
    ;
    # Other instructions don't need ALU
];
