第四题（10分）
考虑如下两个程序（`fact1.c`和`fact2.c`）：

```c
/* fact1.c */
#define MAXNUM 12
int table[MAXNUM];
int fact(int n);
int main(int argc, char **argv) {
    int n;
    table[0] = 0;
    table[1] = 1;
    if (argc == 1) {
        printf("Error: missing argument\n");
        exit(0);
    }
    argv++;
    if (sscanf(*argv, "%d", &n) != 1 || n < 0 || n >= MAXNUM) {
        printf("Error: %s not an int or out of range\n", *argv);
        exit(0);
    }
    printf("fact(%d) = %d\n", n, fact(n));
}
```

```c
/* fact2.c */
int* table;
int fact(int n) {
    static int num = 2;
    if (n >= num) {
        int i = num;
        while (i <= n) {
            table[i] = table[i-1] * i;
            i++;
        }
        num = i;
    }
    return table[n];
}
```
